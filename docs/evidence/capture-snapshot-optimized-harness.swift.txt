import AppKit
import Darwin
import Foundation
import WebKit
import HitSlopCore
import HitSlopCoreBinding
@testable import HitSlopDocument
@testable import HitSlopHost

@main struct CaptureBreakdown {
  @MainActor static func main() async throws {
    _ = NSApplication.shared
    let folder = URL(fileURLWithPath: "/private/tmp/hitslop-capture-spike")
    let root = folder.appendingPathComponent(UUID().uuidString + ".slop")
    try createDocument(template: folder.appendingPathComponent("template.slop").path, destination: root.path)
    defer { try? FileManager.default.removeItem(at: root) }
    let source = try await DocumentSession.open(url: root)
    let initial: [String: Any] = ["title": "Capture breakdown", "tasks": (0..<5000).map {
      ["text": "Task \($0)", "done": false, "archived": false] as [String: Any]
    }]
    let batch = String(decoding: try JSONSerialization.data(withJSONObject: ["intents": [["type": "replace", "path": [], "value": initial]]]), as: UTF8.self)
    _ = try await source.owner.apply(batch: batch)
    try await source.owner.flush()
    source.load()
    try await source.waitUntilReady()
    var records: [[String: Any]] = []
    var pids = Set<Int32>()
    var memory: [[String: Any]] = []
    func footprint(_ pid: Int32) -> UInt64? {
      var usage = rusage_info_v4()
      let result = withUnsafeMutablePointer(to: &usage) { ptr in
        ptr.withMemoryRebound(to: rusage_info_t?.self, capacity: 1) { proc_pid_rusage(pid, RUSAGE_INFO_V4, $0) }
      }
      return result == 0 ? usage.ri_phys_footprint : nil
    }
    func recordMemory(_ sample: Int) {
      let content = pids.compactMap { footprint($0) }
      memory.append(["sample": sample, "hostMiB": Double(footprint(getpid()) ?? 0) / 1_048_576,
        "captureWebContentMiB": Double(content.reduce(0,+)) / 1_048_576, "remainingCaptureProcesses": content.count])
    }
    recordMemory(-1)
    for hiddenEditor in [true] {
      for sample in 0..<21 {
        var phases: [String: Double] = [:]
        var start = Date()
        func mark(_ phase: String) { phases[phase] = Date().timeIntervalSince(start) * 1000; start = Date() }
        let baseline = Date()
        _ = try await SlopRenderer.previewPNGData(session: source)
        let live = Date().timeIntervalSince(baseline) * 1000
        start = Date()
        try await source.flush(); mark("flush")
        let copy = folder.appendingPathComponent(UUID().uuidString + ".slop")
        try await source.owner.copy(to: copy); mark("copy")
        let renderer = try await DocumentSession.open(url: copy, renderTargetsEnabled: true, storage: .snapshot)
        let window = NSWindow(contentRect: renderer.webView.frame, styleMask: [.borderless], backing: .buffered, defer: false)
        window.contentView = renderer.webView
        window.orderOut(nil)
        if hiddenEditor {
          renderer.webView.configuration.userContentController.addUserScript(WKUserScript(source:
            "const s=document.createElement('style');s.textContent='[data-hitslop-root]{display:none!important}';document.head.append(s);",
            injectionTime: .atDocumentEnd, forMainFrameOnly: true))
        }
        mark("open")
        renderer.load()
        try await renderer.waitUntilReady(); mark("ready")
        let key = "_webProcessIdentifier"
        if renderer.webView.responds(to: NSSelectorFromString(key)), let pid = (renderer.webView.value(forKey: key) as? NSNumber)?.int32Value { pids.insert(pid) }
        _ = try await SlopRenderer.previewPNGData(session: renderer); mark("render")
        window.contentView = nil
        try await renderer.close(); mark("close")
        try FileManager.default.removeItem(at: copy); mark("cleanup")
        let total = phases.values.reduce(0, +)
        let record: [String: Any] = ["sample": sample, "hiddenEditor": hiddenEditor, "liveMS": live, "snapshotMS": total, "addedMS": total-live, "phasesMS": phases]
        records.append(record)
        recordMemory(sample)
        print(String(decoding: try JSONSerialization.data(withJSONObject: record, options: [.sortedKeys]), as: UTF8.self))
        fflush(stdout)
      }
    }
    try await source.close()
    try await Task.sleep(for: .seconds(2))
    recordMemory(22)
    let measured = records.dropFirst()
    let paired = measured.map { $0["addedMS"] as! Double }.sorted()
    let added = paired[Int(ceil(Double(paired.count) * 0.95)) - 1]
    let result: [String: Any] = ["rows": 5000, "samples": 20, "warmup": 1,
      "candidate": "saved SQLite copy + fresh snapshot renderer; dedicated export editor root hidden before mount; full 5000-row Export retained",
      "gate": "p95 of paired(snapshot minus live) <= 1000ms", "addedP95MS": added, "latencyGatePassed": added <= 1000,
      "records": records, "memory": memory,
      "memoryScope": "Host plus identified capture WebContent footprints; source WebContent/GPU/network excluded"]
    try JSONSerialization.data(withJSONObject: result, options: [.prettyPrinted, .sortedKeys]).write(to: URL(fileURLWithPath: FileManager.default.currentDirectoryPath).appendingPathComponent("docs/evidence/capture-snapshot-optimized-2026-10-04.json"))
    print("Paired added p95: \(added) ms; gate passed: \(added <= 1000)")
  }
}
